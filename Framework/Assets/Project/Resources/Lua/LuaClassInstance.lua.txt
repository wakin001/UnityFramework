require("Lua/LuaCommonClass")

-- base class
baseClass = BaseClass("Base class", nil)		-- 定义一个基类 base_type
 
function baseClass:__init(x)	-- 定义 base_type 的构造函数
	print("base_type ctor")
	self.x=x
end
 
function baseClass:print_x()	-- 定义一个成员函数 base_type:print_x
	print(baseClass.x)
	print(self.x)
end
 
function baseClass:hello()	-- 定义另一个成员函数 base_type:hello
	print("hello base_type")
end

-- 定义一个类 test 继承于 baseClass
subClass = BaseClass("Inherit class", baseClass)	
 
function subClass:__init()	-- 定义 test 的构造函数
	print("test ctor")
end
 
function subClass:hello()	-- 重载 base_type:hello 为 test:hello
	print("hello test")
end
