function newCounter()
    local i = 0
    return function()
        i = i + 1
        return i
    end
end

local fact
fact = function(n)
    if n == 0 then
        return 1
    else
        return n * fact(n - 1) --- 要事先声明local fact，否则回去找global的fact
    end
end

--- start 协程 ---------
local count = 0
local function receive(prod)
    local status, value = coroutine.resume(prod)
    return value
end

local function send(x)
    coroutine.yield(x)
end

local function producer()
    return coroutine.create(
        function()
            while count < 20 do
                count = count + 1
                print("produce one : ", count)
                send(count)
            end
        end
    )
end

local function consumer(prod)
    while count < 20 do
        local x = receive(prod) -- get new value
        print("consume one: ", x) -- consume new value end
    end
end
