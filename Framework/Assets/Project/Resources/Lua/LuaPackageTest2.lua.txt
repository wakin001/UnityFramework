local P = {} -- package
if _REQUIREDNAME == nil then
    print("_REQUIREDNAME == nil")
    complex = P
else
    print("_REQUIREDNAME != nil")
    _G[_REQUIREDNAME] = P
end
setfenv(1, P)

function new(r, i)
    return {r = r, i = i}
end

-- constant
i = new(0, 1)

function checkComplex(c)
    if not ((type(c) == "table") and tonumber(c.r) and tonumber(c.i)) then
        error("bad complex number", 3)
    end
end

function add(c1, c2)
    checkComplex(c1)
    checkComplex(c2)
    return new(c1.r + c2.r, c1.i + c2.i)
end

function mul(c1, c2)
    return new(c1.r * c2.r - c1.i * c2.i, c1.r * c2.i + c1.i * c2.r)
end

function inv(c)
    local n = c.r ^ 2 + c.i ^ 2
    return new(c.r / n, -c.i / n)
end

function sub(c1, c2)
    return new(c1.r - c2.r, c1.i - c2.i)
end

return complex
