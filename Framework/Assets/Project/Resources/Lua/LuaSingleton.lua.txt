--保存类类型的虚表

require("Lua/LuaCommonClass")

SingletonBase = BaseClass("SingletonBase")
SingletonBase.instance = nil
SingletonBase.GetInstnace = function(...)
    if SingletonBase.instance == nil then
        SingletonBase.instance = SingletonBase.New(...)
    end
    return SingletonBase.instance
end

SingletonBase.__init = function()
end
